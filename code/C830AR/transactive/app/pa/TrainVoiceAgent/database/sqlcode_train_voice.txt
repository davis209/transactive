TRAINVOICE_SELECT_97001=SELECT t.train_id,t.train_name,COALESCE(c.car_id,0) AS car_id,COALESCE(c.car_name,'') AS car_name,COALESCE(c.car_position,0) AS car_position,COALESCE(d.device_key,0) AS device_key,COALESCE(d.device_type,'') AS device_type,COALESCE(d.device_name,'') AS device_name,COALESCE(d.protocol_device_id,0) AS protocol_device_id,COALESCE(d.ip_address,'') AS ip_address,COALESCE(d.control_port,0) AS control_port,COALESCE(d.sip_extension,'') AS sip_extension FROM train_voice_train t LEFT JOIN train_voice_car c ON c.train_id=t.train_id AND c.enabled=1 LEFT JOIN train_voice_device d ON d.car_id=c.car_id AND d.enabled=1 WHERE t.enabled=1 ORDER BY t.train_id,c.car_position,d.device_key
